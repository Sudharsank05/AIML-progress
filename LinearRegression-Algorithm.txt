import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score

# 1. Data
X = np.array([[20], [25], [30], [35], [40], [45]])
y = np.array([100, 150, 200, 250, 300, 350])

# 2. Model Train
model = LinearRegression()
model.fit(X, y)

# 3. Coefficients
print(f"Slope (m): {model.coef_[0]}")
print(f"Intercept (c): {model.intercept_}")
print(f"Equation: y = {model.coef_[0]}x + {model.intercept_}")

# 4. Test Data
X_test = np.array([[50], [55], [60]])
y_test = np.array([400, 450, 500])  # Actual values

# 5. Predict
predictions = model.predict(X_test)
print(f"\nPredictions:")
for i, x in enumerate(X_test):
    print(f"  {x[0]}°C → ₹{predictions[i]:.2f} (Actual: ₹{y_test[i]})")

# 6. Evaluate
mse = mean_squared_error(y_test, predictions)
r2 = r2_score(y_test, predictions)
print(f"\nMSE: {mse:.2f}")
print(f"R²: {r2:.4f}")

# 7. Visualize
plt.scatter(X, y, color='blue', label='Training Data')
plt.plot(X, model.predict(X), color='red', label='Regression Line')
plt.scatter(X_test, y_test, color='green', label='Test Data', marker='^')
plt.scatter(X_test, predictions, color='orange', label='Predictions', marker='x')
plt.xlabel('Temperature (°C)')
plt.ylabel('Ice Cream Sales (₹)')
plt.legend()
plt.title('Linear Regression: Temperature vs Sales')
plt.show()
